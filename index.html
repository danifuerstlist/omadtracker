// ===== TRACKER =====

let trackerData = JSON.parse(localStorage.getItem('fastenTracker') || '{}');

let weightData = JSON.parse(localStorage.getItem('fastenWeight') || '[]');

let currentModalDate = null;

let selectedType = null;


const today = new Date();

today.setHours(0,0,0,0);


function buildTracker() {

  const container = document.getElementById('tracker-container');

  container.innerHTML = '';

  

  weekPlans.forEach((w, wi) => {

    const wrap = document.createElement('div');

    wrap.className = 'tracker-week';

    

    const title = document.createElement('div');

    title.className = 'tracker-week-title';

    title.textContent = `Woche ${wi+1} — ${w.label}`;

    wrap.appendChild(title);

    

    const grid = document.createElement('div');

    grid.className = 'tracker-days';

    

    w.days.forEach((d, di) => {

      const dateKey = `w${wi}d${di}`;

      const dayDiv = document.createElement('div');

      dayDiv.className = 'tracker-day';

      

      const planned = weekPlans[wi].days[di];

      // Get actual date for this slot

      const dayOffset = wi * 7 + di;

      const thisDate = new Date(START);

      thisDate.setDate(START.getDate() + dayOffset);

      thisDate.setHours(0,0,0,0);

      const isFuture = thisDate > today;

      const shortName = d.name.split(' ')[0];

      

      if (trackerData[dateKey]) {

        const t = trackerData[dateKey];

        dayDiv.classList.add(`done-${t}`);

        dayDiv.innerHTML = `<span>${t === 'omad' ? '⚡' : t === '2meal' ? '🌙' : '✓'}</span>${shortName}`;

      } else if (isFuture) {

        dayDiv.classList.add('future');

        dayDiv.innerHTML = `<span>${typeIcon(planned.type)}</span>${shortName}`;

      } else {

        dayDiv.innerHTML = `<span>○</span>${shortName}`;

        dayDiv.onclick = () => openModal(dateKey, d.name);

      }

      

      grid.appendChild(dayDiv);

    });

    

    wrap.appendChild(grid);

    container.appendChild(wrap);

  });

  

  updateStats();

}


function openModal(dateKey, label) {

  currentModalDate = dateKey;

  selectedType = null;

  document.getElementById('modal-date').textContent = label;

  document.getElementById('btn-omad').className = 'select-btn';

  document.getElementById('btn-2meal').className = 'select-btn';

  document.getElementById('btn-normal').className = 'select-btn';

  document.getElementById('modal').classList.add('open');

}


function closeModal() {

  document.getElementById('modal').classList.remove('open');

  currentModalDate = null;

}


function selectType(type) {

  selectedType = type;

  document.getElementById('btn-omad').className = 'select-btn' + (type === 'omad' ? ' sel-omad' : '');

  document.getElementById('btn-2meal').className = 'select-btn' + (type === '2meal' ? ' sel-2meal' : '');

  document.getElementById('btn-normal').className = 'select-btn' + (type === 'normal' ? ' sel-normal' : '');

}


function saveDay() {

  if (!selectedType || !currentModalDate) return;

  trackerData[currentModalDate] = selectedType;

  localStorage.setItem('fastenTracker', JSON.stringify(trackerData));

  closeModal();

  buildTracker();

}


// ===== WEIGHT =====

function logWeight() {

  const val = parseFloat(document.getElementById('weight-input').value);

  if (isNaN(val) || val < 30 || val > 200) return;

  const now = new Date();

  const entry = { kg: val, date: now.toLocaleDateString('de-DE', { day:'2-digit', month:'2-digit' }), ts: Date.now() };

  weightData.unshift(entry);

  localStorage.setItem('fastenWeight', JSON.stringify(weightData));

  document.getElementById('weight-input').value = '';

  renderWeight();

  updateStats();

}


function renderWeight() {

  const log = document.getElementById('weight-log');

  if (weightData.length === 0) {

    log.innerHTML = '<div class="weight-entry" style="justify-content:center; color:var(--muted); font-size:11px; padding:20px;">Noch keine Einträge</div>';

    return;

  }

  log.innerHTML = weightData.map((e, i) => {

    const prev = weightData[i+1];

    let diffHtml = '';

    if (prev) {

      const diff = e.kg - prev.kg;

      const sign = diff > 0 ? '+' : '';

      diffHtml = `<span class="weight-diff ${diff <= 0 ? 'diff-down' : 'diff-up'}">${sign}${diff.toFixed(1)} kg</span>`;

    }

    return `<div class="weight-entry">

      <span style="color:var(--muted)">${e.date}</span>

      <span class="weight-kg">${e.kg.toFixed(1)} kg</span>

      ${diffHtml}

    </div>`;

  }).join('');

  

  // Progress

  const current = weightData[0].kg;

  const lost = 63 - current;

  const pct = Math.max(0, Math.min(100, (lost / 5) * 100));

  document.getElementById('progress-fill').style.width = pct + '%';

  document.getElementById('progress-pct').textContent = Math.round(pct) + '%';

}


function updateStats() {

  // Current weight

  const cw = weightData.length > 0 ? weightData[0].kg : 63.0;

  document.getElementById('stat-weight').textContent = cw.toFixed(1);

  const lost = (63 - cw).toFixed(1);

  document.getElementById('stat-lost').textContent = lost > 0 ? lost : '0.0';

  

  // OMAD count

  const omadCount = Object.values(trackerData).filter(v => v === 'omad').length;

  document.getElementById('stat-omad').textContent = omadCount;

  

  // Current week

  const daysSinceStart = Math.floor((today - START) / 86400000);

  const weekNum = Math.min(10, Math.floor(daysSinceStart / 7) + 1);

  document.getElementById('stat-week').textContent = `W${weekNum}`;

}


function showTab(name) {

  document.querySelectorAll('.tab').forEach((t, i) => {

    const tabs = ['plan','tracker','weight','mindset'];

    t.classList.toggle('active', tabs[i] === name);

  });

  document.querySelectorAll('.section').forEach(s => s.classList.remove('active'));

  document.getElementById('tab-' + name).classList.add('active');

}
